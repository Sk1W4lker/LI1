Função principal

jogoTerminou :: Jogo -> Bool Source #

Indica se o jogador perdeu o jogo, o que acontece quando está fora do mapa, dentro de água ou na mesma posição de um carro.

>>> jogoTerminou (Jogo (Jogador (0,0)) (Mapa 2 [(Rio 1, [Nenhum, Tronco])]))
True
>>> jogoTerminou (Jogo (Jogador (1,0)) (Mapa 2 [(Rio 1, [Nenhum, Tronco])]))
False
